📊 Conectar con tu Google Sheets

Seguí estos pasos para que la app guarde los datos directamente en tu planilla de Drive.

✅ Google Sheets conectado. Los registros nuevos se guardan automáticamente en tu planilla.
1Abrí tu Google Sheet En tu planilla de Drive, andá al menú: Extensiones → Apps Script
2Pegá este código (reemplazá todo) Borrá lo que haya en el editor y pegá este script:
function guardarDatos(data) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(data.sheet || "Hoja1"); if (!sheet) sheet = ss.getActiveSheet(); var dni = String(data.row[5] || "").trim(); var found = false; if (dni) { var lastRow = sheet.getLastRow(); for (var i = 6; i <= lastRow; i++) { var cellDni = String(sheet.getRange(i, 6).getValue()).trim(); if (cellDni === dni) { sheet.getRange(i, 1, 1, data.row.length).setValues([data.row]); found = true; break; } } } if (!found) sheet.appendRow(data.row); return found; } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var found = guardarDatos(data); return ContentService .createTextOutput(JSON.stringify({status:"ok", updated: found})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({status:"error", msg:err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { // Si viene con datos (desde archivo local via GET) if (e.parameter && e.parameter.data) { var decoded = decodeURIComponent(escape(atob(e.parameter.data))); var data = JSON.parse(decoded); var found = guardarDatos(data); return ContentService .createTextOutput(JSON.stringify({status:"ok", updated: found})) .setMimeType(ContentService.MimeType.JSON); } // Si es solo un ping de prueba return ContentService .createTextOutput(JSON.stringify({status:"ok", msg:"Script activo"})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({status:"error", msg:err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
